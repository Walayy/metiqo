async function captureMarketPage(name){const data=await captureFullStake(name);const markets=await stakeTab.playwright.evaluate(()=>[...document.querySelectorAll('.groups .secondary-accordion')].map((m,i)=>({order:i,title:m.querySelector('.header')?.innerText.split('\n')[0],expanded:m.classList.contains('is-open'),text:m.innerText,outcomes:[...m.querySelectorAll('[data-testid="fixture-outcome"]')].map((b,j)=>({order:j,label:b.getAttribute('aria-label'),name:b.querySelector('[data-testid="outcome-button-name"]')?.innerText??null,odds:b.querySelector('[data-testid="fixture-odds"]')?.innerText??null,disabled:b.hasAttribute('disabled'),text:b.innerText})),controls:[...m.querySelectorAll('button')].filter(b=>b.getAttribute('data-testid')!=='fixture-outcome').map(b=>({text:b.innerText,label:b.getAttribute('aria-label'),disabled:b.hasAttribute('disabled')}))})));const record={capture:name,observedAt:auditEvidence.at(-1).observedAt,url:data.url,markets};marketEvidence.push(record);await auditFs.writeFile(auditDir+'/markets.json',JSON.stringify(marketEvidence,null,2));nodeRepl.write({name,markets:markets.map(m=>m.title),selections:markets.reduce((s,m)=>s+m.outcomes.length,0)});return record;}
