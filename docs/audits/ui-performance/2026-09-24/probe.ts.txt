// Temporary, local-only instrumentation used for the before/after audit.
let commits = 0;
let renderMs = 0;
export function auditRender(_id: string, _phase: string, duration: number) {
  commits++;
  renderMs += duration;
}
if (import.meta.env.DEV) {
  const button = document.createElement('button');
  button.textContent = 'Mesurer 12 s';
  button.style.cssText = 'position:fixed;bottom:8px;left:8px;z-index:99999;background:#fff;color:#111;padding:8px';
  document.body.append(button);
  button.onclick = () => {
    commits = 0;
    renderMs = 0;
    const frames: number[] = [];
    const tasks: number[] = [];
    let previous = performance.now();
    const start = previous;
    const observer = new PerformanceObserver((list) => {
      tasks.push(...list.getEntries().map((entry) => entry.duration));
    });
    observer.observe({ type: 'longtask' });
    button.textContent = 'Mesure en cours';
    const tick = (now: number) => {
      frames.push(now - previous);
      previous = now;
      if (now - start < 12000) requestAnimationFrame(tick);
      else {
        observer.disconnect();
        frames.sort((a, b) => a - b);
        const result = { commits, renderMs: Math.round(renderMs), frames: frames.length,
          p95: frames[Math.floor(frames.length * .95)], over33ms: frames.filter((n) => n > 33.4).length,
          longTasks: tasks.length, longTaskMs: Math.round(tasks.reduce((a,b)=>a+b,0)) };
        button.textContent = 'Mesurer 12 s';
        let output = document.querySelector('#perf-result');
        if (!output) { output = document.createElement('output'); output.id='perf-result'; output.setAttribute('hidden',''); document.body.append(output); }
        output.textContent = JSON.stringify(result);
      }
    };
    requestAnimationFrame(tick);
  };
}
